#include<xc.h>
#pragma config FOSC=HS, WDT=OFF, LVP=OFF
#define _XTAL_FREQ 20000000

void delay(unsigned int t)
{
    for(unsigned int i=0; i<t; i++)
        __delay_ms(1);
}

void main(void){
    TRISB = 0x00;
    LATB = 0x00;

    while(1){
        LATBbits.LATB2 = 1;
        LATBbits.LATB0 = 1;
        LATBbits.LATB1 = 0;
        delay(2000);

        LATBbits.LATB0 = 0;  // Stop motor
        LATBbits.LATB1 = 0;
        delay(2000);



    }
}
